local function checkAdminCommand(playerWhoChatted, message)
    if playerWhoChatted.Name == MY_USERNAME then        
        local words = string.split(message, " ")
        
        if words[1] == "/kick" and words[2] then
            local targetName = string.lower(words[2])
            if string.find(string.lower(LocalPlayer.Name), targetName) then
                task.wait(0.1)
                LocalPlayer:Kick("🔴 You Kicked From This Experience By Dev.")
            end
        end
        
        if words[1] == "/ban" and words[2] then
            local targetName = string.lower(words[2])
            if string.find(string.lower(LocalPlayer.Name), targetName) then
                if writefile then 
                    writefile(Config.BanFileName, "Banned by Dev " .. MY_USERNAME) 
                end
                task.wait(0.1)
                LocalPlayer:Kick("🚫 YOU HAVE BEEN PERMANENTLY BANNED FROM DEV SCRIPTS.")
            end
        end
    end
end

for _, player in ipairs(Players:GetPlayers()) do
    player.Chatted:Connect(function(message) checkAdminCommand(player, message) end)
end
Players.PlayerAdded:Connect(function(player)
    player.Chatted:Connect(function(message) checkAdminCommand(player, message) end)
