-- [[ FUNCTIONS ]]

local function appendChat(sender, message)
    local TextLabel = Instance.new("TextLabel")
    TextLabel.Size = UDim2.new(1, 0, 0, 20)
    TextLabel.BackgroundTransparency = 1
    TextLabel.Text = string.format("[%s]: %s", sender, message)
    TextLabel.TextColor3 = (sender == LocalPlayer.Name) and Color3.fromRGB(100, 200, 255) or Color3.fromRGB(255, 255, 255)
    TextLabel.TextSize = 14
    TextLabel.Font = Enum.Font.SourceSans
    TextLabel.TextXAlignment = Enum.TextXAlignment.Left
    TextLabel.Parent = ScrollingFrame
    
    ScrollingFrame.CanvasPosition = Vector2.new(0, ScrollingFrame.AbsoluteWindowSize.Y * 10)
end

local function sendToDiscord(msg)
    -- Safety Check: Kung walang nakitang valid request function ang executor, mag-wa-warn lang imbes na mag-crash
    if not custom_request then 
        warn("[Private Chat Error]: Ang executor mo ay hindi sumusuporta sa HTTP Requests / Webhooks.")
        return 
    end
    
    local data = {
        ["username"] = LocalPlayer.Name .. " (In-Game)",
        ["content"] = msg
    }
    
    -- Pinalitan ng pcall para hindi mag-error ang laro kahit mag-fail ang discord ping
    pcall(function()
        custom_request({
            Url = WEBHOOK_URL,
            Method = "POST",
            Headers = {
                ["Content-Type"] = "application/json"
            },
            Body = HttpService:JSONEncode(data)
        })
    end)
end

-- Pagsend kapag clinick ang button o pinindot ang Enter
local function onSend()
    local text = TextBox.Text
    if text ~= "" then
        TextBox.Text = ""
        appendChat(LocalPlayer.Name, text)
        sendToDiscord(text)
    end
end

SendButton.MouseButton1Click:Connect(onSend)
TextBox.FocusLost:Connect(function(enterPressed)
    if enterPressed then onSend() end
end)
